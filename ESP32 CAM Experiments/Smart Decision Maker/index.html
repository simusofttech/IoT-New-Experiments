Working Principle

Browse activities, shortlist choices and randomly select one using three buttons. The controller reads inputs, processes them in the sketch and updates outputs.

Procedure

  1. Verify the components and assemble the circuit using the connection table.
  2. Install the ESP32 board support and required Arduino libraries.
  3. Open the sketch, check board and GPIO assignments and configure credentials where applicable.
  4. Compile and upload the program; open Serial Monitor at 115200 baud.
  5. Test each input and verify the display and outputs.

#include <Arduino.h>
#include <Wire.h>
#include <U8g2lib.h>
U8G2_SSD1306_128X64_NONAME_F_HW_I2C oled(U8G2_R0,U8X8_PIN_NONE);
const int NEXT=18, SELECT=19, DECIDE=23, GREEN=25, RED=26;
const char* options[]={"Study","Exercise","Read a book","Take a break","Learn coding","Go for a walk"};
const int COUNT=6;
bool picked[COUNT]={false};
int current=0, count=0;
bool lastN=HIGH,lastS=HIGH,lastD=HIGH;
unsigned long lastPress=0;
void screen(const String &msg="") {
 oled.clearBuffer(); oled.setFont(u8g2_font_6x10_tf);
 oled.drawStr(0,12,"SMART DECISION");
 if(msg.length()) oled.drawStr(0,30,msg.c_str());
 else { oled.drawStr(0,30,options[current]); oled.drawStr(0,46,picked[current]?"[ADDED]":"NEXT / SELECT / DECIDE"); }
 oled.sendBuffer();
}
void setup(){
 Serial.begin(115200); Wire.begin(21,22); oled.begin();
 pinMode(NEXT,INPUT_PULLUP);pinMode(SELECT,INPUT_PULLUP);pinMode(DECIDE,INPUT_PULLUP);
 pinMode(GREEN,OUTPUT);pinMode(RED,OUTPUT);randomSeed(micros());screen();
}
void loop(){
 bool n=digitalRead(NEXT),s=digitalRead(SELECT),d=digitalRead(DECIDE);
 if(millis()-lastPress>60){
  if(lastN && !n){current=(current+1)%COUNT;digitalWrite(GREEN,LOW);screen();lastPress=millis();}
  else if(lastS && !s){if(!picked[current]){picked[current]=true;count++;digitalWrite(RED,HIGH);Serial.println(options[current]);screen("Option added!");}else screen("Already selected!");digitalWrite(GREEN,LOW);lastPress=millis();}

 }
 lastN=n;lastS=s;lastD=d;
}
