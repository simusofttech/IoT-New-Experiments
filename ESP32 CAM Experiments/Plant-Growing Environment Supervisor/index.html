Working Principle

Monitor temperature, humidity and light with DHT22 and a digital LDR, show alerts on an OLED, and upload readings to ThingSpeak. The controller reads inputs, processes them in the sketch and updates outputs.

Procedure

  1. Verify the components and assemble the circuit using the connection table.
  2. Install the ESP32 board support and required Arduino libraries.
  3. Open the sketch, check board and GPIO assignments and configure credentials where applicable.
  4. Compile and upload the program; open Serial Monitor at 115200 baud.
  5. Test each input and verify the display and outputs.

#include <WiFi.h>
#include <ThingSpeak.h>
#include <Wire.h>
#include <U8g2lib.h>
#include <DHT.h>
const char* SSID="YOUR_WIFI_SSID";
const char* PASSWORD="YOUR_WIFI_PASSWORD";
unsigned long CHANNEL_ID=1234567;
const char* API_KEY="YOUR_WRITE_API_KEY";
WiFiClient client;
DHT dht(16,DHT22);
U8G2_SSD1306_128X64_NONAME_F_HW_I2C oled(U8G2_R0,U8X8_PIN_NONE);
const int LDR=34,GREEN=25,RED=26,BUZZER=27;
unsigned long lastRead=0,lastSend=0;
float temp=NAN,hum=NAN;
int light=0,alert=5;
void readSensors(){
 temp=dht.readTemperature();hum=dht.readHumidity();
 light=digitalRead(LDR)==LOW?100:0;
 if(isnan(temp)||isnan(hum))alert=5;
 else {
  bool t=temp<18||temp>30,h=hum<40||hum>80,l=light==0;
  int count=t+h+l;
  alert=count>1?4:t?1:h?2:l?3:0;
 }
 digitalWrite(GREEN,alert==0);
 digitalWrite(RED,alert!=0);
 digitalWrite(BUZZER,alert!=0);
 oled.clearBuffer();oled.setFont(u8g2_font_6x10_tf);
 oled.drawStr(0,10,"PLANT SUPERVISOR");
 char line[32];
 if(isnan(temp))snprintf(line,sizeof(line),"Temp: SENSOR ERROR");
 else snprintf(line,sizeof(line),"Temp: %.1f C",temp);
 oled.drawStr(0,25,line);
 if(isnan(hum))snprintf(line,sizeof(line),"Humidity: ERROR");
 else snprintf(line,sizeof(line),"Humidity: %.1f %%",hum);
 oled.drawStr(0,39,line);
 oled.drawStr(0,51,light?"Light: LIGHT":"Light: DARK");
 snprintf(line,sizeof(line),"Alert code: %d",alert);oled.drawStr(0,63,line);
 oled.sendBuffer();
 Serial.printf("Temp %.1f, Hum %.1f, Light %d, Alert %d\n",temp,hum,light,alert);
}
void setup(){
 Serial.begin(115200);pinMode(LDR,INPUT);
 pinMode(GREEN,OUTPUT);pinMode(RED,OUTPUT);pinMode(BUZZER,OUTPUT);
 Wire.begin(21,22);oled.begin();dht.begin();
 WiFi.mode(WIFI_STA);WiFi.begin(SSID,PASSWORD);
 ThingSpeak.begin(client);readSensors();
 lastRead=millis();lastSend=millis();
}
void loop(){
 unsigned long now=millis();
 if(now-lastRead>=2500){lastRead=now;readSensors();}
 if(now-lastSend>=20000){
  lastSend=now;
  if(WiFi.status()==WL_CONNECTED && !isnan(temp) && !isnan(hum)){
   ThingSpeak.setField(1,temp);ThingSpeak.setField(2,hum);
   ThingSpeak.setField(3,light);ThingSpeak.setField(4,alert);
   Serial.println(ThingSpeak.writeFields(CHANNEL_ID,API_KEY));
  }
 }
 if(WiFi.status()!=WL_CONNECTED){
  static unsigned long retry=0;
  if(now-retry>15000){retry=now;WiFi.disconnect();WiFi.begin(SSID,PASSWORD);}
 }
}

Expected Output

OLED shows temperature, humidity, light status and alert code. Green LED indicates normal conditions; red LED and buzzer indicate an alert. Valid readings are uploaded to ThingSpeak every 20 seconds.

Result

The experiment demonstrates sensor/input processing, display feedback and controller output.

What You Learn

GPIO interfacing, digital inputs and outputs, embedded programming and real-time feedback.
