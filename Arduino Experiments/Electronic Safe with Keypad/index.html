Electronic Safe with Keypad

← Arduino Experiments Aim Components Connections Working Procedure Code Result

Aim

To build an electronic safe using Arduino UNO, a 4×4 keypad, servo motor, 16×2 I²C LCD and buzzer, where the safe unlocks only when the correct four-digit PIN is entered.

About the Experiment

The user enters a four-digit PIN using the keypad and presses # to verify it. If the entered PIN is correct, the servo moves from the locked position to the unlocked position, the LCD shows access granted, and the buzzer gives a success sound.

If the PIN is incorrect, the LCD shows access denied and the buzzer gives an error sound. When the safe is unlocked, pressing * locks it again.

Circuit Diagram

Connect keypad rows R1–R4 to D2–D5 and columns C1–C4 to D6–D9. Connect the servo signal to D10, buzzer positive to D11, and the I²C LCD to A4 and A5.

Procedure

  1. Connect keypad rows R1 to R4 to Arduino pins D2 to D5.
  2. Connect keypad columns C1 to C4 to Arduino pins D6 to D9.
  3. Connect the servo signal wire to D10.
  4. Connect the servo VCC to 5V and GND to GND.
  5. Connect buzzer positive to D11 and buzzer negative to GND.
  6. Connect LCD VCC to 5V and LCD GND to GND.
  7. Connect LCD SDA to A4 and SCL to A5.
  8. Connect Arduino UNO to the computer.
  9. Open Arduino IDE.
  10. Install the Keypad, Servo and LiquidCrystal_I2C libraries if required.
  11. Enter the program given below.
  12. Select Arduino UNO and the correct COM port.
  13. Verify and upload the program.
  14. Enter the PIN using the keypad.
  15. Press # to check the PIN.
  16. If unlocked, press * to lock the safe again.

Arduino Code

Electronic_Safe_with_Keypad.ino 
#include <Wire.h>
#include <LiquidCrystal_I2C.h>
#include <Keypad.h>
#include <Servo.h>

// LCD
LiquidCrystal_I2C lcd(0x27, 16, 2);

// SERVO
Servo lockServo;

#define SERVO_PIN 10

const int LOCKED_POSITION = 0;
const int UNLOCKED_POSITION = 90;

// BUZZER
#define BUZZER_PIN 11

// KEYPAD
const byte ROWS = 4;
const byte COLS = 4;

char keys[ROWS][COLS] = {
  {'1', '2', '3', 'A'},
  {'4', '5', '6', 'B'},
  {'7', '8', '9', 'C'},
  {'*', '0', '#', 'D'}
};

byte rowPins[ROWS] = {2, 3, 4, 5};
byte colPins[COLS] = {6, 7, 8, 9};

Keypad keypad = Keypad(
  makeKeymap(keys),
  rowPins,
  colPins,
  ROWS,
  COLS
);

// PIN SETTINGS
const char correctPIN[] = "1234";
const byte PIN_LENGTH = 4;

char enteredPIN[PIN_LENGTH + 1];
byte enteredLength = 0;

// SAFE STATUS
bool safeUnlocked = false;

void setup() {

  Serial.begin(9600);

  lcd.init();
  lcd.backlight();

  lockServo.attach(SERVO_PIN);
  lockServo.write(LOCKED_POSITION);

  pinMode(BUZZER_PIN, OUTPUT);
  digitalWrite(BUZZER_PIN, LOW);

  lcd.clear();
  lcd.setCursor(0, 0);
  lcd.print(" ELECTRONIC SAFE");

  lcd.setCursor(0, 1);
  lcd.print(" System Starting");

  delay(2000);

  showEnterPIN();

  Serial.println("============================");
  Serial.println(" ELECTRONIC SAFE SYSTEM");
  Serial.println("============================");
  Serial.println("Safe Locked");
  Serial.println("Enter PIN and press #");
}

void loop() {

  char key = keypad.getKey();

  if (key) {

    Serial.print("Key Pressed: ");
    Serial.println(key);

    // Clear PIN or lock safe
    if (key == '*') {

      if (safeUnlocked) {
        lockSafe();
      } else {
        clearPIN();
        showEnterPIN();
      }
    }

    // Check PIN
    else if (key == '#') {
      checkPIN();
    }

    // Enter number
    else if (key >= '0' && key <= '9') {
      enterDigit(key);
    }

    // Ignore A, B, C, D
    else {
      // No action
    }
  }
}

void enterDigit(char key) {

  if (safeUnlocked) {
    return;
  }

  if (enteredLength >= PIN_LENGTH) {

    lcd.setCursor(0, 1);
    lcd.print("Press # to check");

    return;
  }

  enteredPIN[enteredLength] = key;
  enteredLength++;
  enteredPIN[enteredLength] = '\0';

  lcd.setCursor(0, 1);

  for (byte i = 0; i < enteredLength; i++) {
    lcd.print('*');
  }

  for (byte i = enteredLength; i < 16; i++) {
    lcd.print(' ');
  }

  Serial.print("Entered digits: ");
  Serial.println(enteredLength);
}

void checkPIN() {

  if (safeUnlocked) {
    return;
  }

  if (enteredLength != PIN_LENGTH) {

    lcd.clear();

    lcd.setCursor(0, 0);
    lcd.print("Enter 4 Digits");

    lcd.setCursor(0, 1);
    lcd.print("Then press #");

    errorBeep();

    delay(1500);

    showEnterPIN();

    return;
  }

  enteredPIN[enteredLength] = '\0';

  if (strcmp(enteredPIN, correctPIN) == 0) {

    unlockSafe();

  } else {

    wrongPIN();
  }
}

void unlockSafe() {

  safeUnlocked = true;

  lcd.clear();

  lcd.setCursor(0, 0);
  lcd.print(" ACCESS GRANTED");

  lcd.setCursor(0, 1);
  lcd.print("Safe Unlocked");

  Serial.println("Correct PIN!");
  Serial.println("Access Granted");
  Serial.println("Servo: UNLOCKED");

  lockServo.write(UNLOCKED_POSITION);

  successBeep();

  delay(2000);

  lcd.clear();

  lcd.setCursor(0, 0);
  lcd.print("Safe Open");

  lcd.setCursor(0, 1);
  lcd.print("Press * to Lock");
}

void lockSafe() {

  safeUnlocked = false;

  lockServo.write(LOCKED_POSITION);

  lcd.clear();

  lcd.setCursor(0, 0);
  lcd.print("Safe Locked");

  lcd.setCursor(0, 1);
  lcd.print("Lock Successful");

  Serial.println("Safe Locked");

  tone(BUZZER_PIN, 1000, 150);

  delay(1500);

  clearPIN();

  showEnterPIN();
}

void wrongPIN() {

  lcd.clear();

  lcd.setCursor(0, 0);
  lcd.print(" WRONG PIN!");

  lcd.setCursor(0, 1);
  lcd.print("Access Denied");

  Serial.println("Wrong PIN!");
  Serial.println("Access Denied");

  errorBeep();

  delay(2000);

  clearPIN();

  showEnterPIN();
}

void showEnterPIN() {

  lcd.clear();

  lcd.setCursor(0, 0);
  lcd.print("Enter PIN:");

  lcd.setCursor(0, 1);

  for (byte i = 0; i < enteredLength; i++) {
    lcd.print('*');
  }
}

void clearPIN() {

  enteredLength = 0;

  for (byte i = 0; i <= PIN_LENGTH; i++) {
    enteredPIN[i] = '\0';
  }
}

void successBeep() {

  tone(BUZZER_PIN, 1500, 150);
  delay(200);

  tone(BUZZER_PIN, 2000, 200);
  delay(250);

  noTone(BUZZER_PIN);
}

void errorBeep() {

  for (byte i = 0; i < 3; i++) {

    tone(BUZZER_PIN, 400, 150);

    delay(200);
  }

  noTone(BUZZER_PIN);
}

Expected Output

The LCD asks the user to enter a four-digit PIN. Each entered digit is shown as *. Pressing # verifies the PIN.

Correct PIN: Access Granted, servo moves to 90°
Wrong PIN: Access Denied, error buzzer sounds
Lock Again: Press * while safe is open
Default PIN: 1234

Result

The Electronic Safe with Keypad experiment was implemented using Arduino UNO, a 4×4 keypad, servo motor, 16×2 I²C LCD and buzzer. The safe unlocks only when the correct PIN is entered and locks again when * is pressed.
