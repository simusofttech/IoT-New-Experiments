Stepper Motor Position Selector

← Arduino Experiments Aim Components Connections Working Controls Procedure Code Result

Aim

To design and operate a keypad-controlled motor position selector in Tinkercad, where the user enters a numerical target position using a 4×4 keypad and Arduino drives the motor toward the selected position through an L293D motor driver.

About the Experiment

The user enters a target position using the 4×4 keypad. Arduino processes the entered value, displays the target and current position on the 16×2 I²C LCD, and controls the L293D motor driver to move the motor forward or reverse.

The keypad also provides control functions for clearing input, stopping the motor, showing the current position, zeroing the position, and resetting the position counter.

Working Principle

Arduino Starts
↓
LCD shows target and current position
↓
User enters a numerical target using keypad
↓
Press # to confirm target
↓
Arduino compares target with current position
↓
Target greater → Motor moves forward
Target smaller → Motor moves reverse
↓
Current position changes step by step
↓
Target reached
↓
Motor stops and LCD shows final position
Position Update: One position unit per simulation step
Step Delay: 100 ms
LCD: Shows Target and Current position

Procedure

  1. Connect keypad rows R1–R4 to Arduino D4–D7.
  2. Connect keypad columns C1–C4 to Arduino D8–D11.
  3. Connect LCD SDA to A4 and SCL to A5.
  4. Connect L293D EN1 to D12.
  5. Connect L293D IN1 to D13.
  6. Connect L293D IN2 to A0.
  7. Connect the motor to L293D OUT1 and OUT2.
  8. Connect L293D logic supply to 5V and common ground to GND.
  9. Provide a suitable motor supply.
  10. Open the circuit in Tinkercad.
  11. Enter the Arduino code given below.
  12. Start the simulation.
  13. Enter a target position using the keypad.
  14. Press # to start movement toward the target.
  15. Use *, A, B, C and D for the available control functions.
  16. Observe target and current position on the LCD.

Arduino Code

Stepper_Motor_Position_Selector.ino 
#include <Keypad.h>
#include <Wire.h>
#include <LiquidCrystal_I2C.h>

LiquidCrystal_I2C lcd(0x27, 16, 2);

// KEYPAD
const byte ROWS = 4;
const byte COLS = 4;

char keys[ROWS][COLS] =
{
  {'1','2','3','A'},
  {'4','5','6','B'},
  {'7','8','9','C'},
  {'*','0','#','D'}
};

byte rowPins[ROWS] = {4, 5, 6, 7};
byte colPins[COLS] = {8, 9, 10, 11};

Keypad keypad = Keypad(
  makeKeymap(keys),
  rowPins,
  colPins,
  ROWS,
  COLS
);

// MOTOR DRIVER
const int EN1 = 12;
const int IN1 = 13;
const int IN2 = A0;

// POSITION
long currentPosition = 0;
long targetPosition = 0;

String inputNumber = "";

void setup()
{
  pinMode(EN1, OUTPUT);
  pinMode(IN1, OUTPUT);
  pinMode(IN2, OUTPUT);

  motorStop();

  lcd.init();
  lcd.backlight();

  lcd.setCursor(0, 0);
  lcd.print("POSITION SELECT");

  lcd.setCursor(0, 1);
  lcd.print("Enter Target");

  delay(2000);

  showTarget();
}

void loop()
{
  char key = keypad.getKey();

  if (key)
  {
    // Numerical target entry
    if (key >= '0' && key <= '9')
    {
      if (inputNumber.length() < 6)
      {
        inputNumber += key;

        lcd.clear();
        lcd.setCursor(0, 0);
        lcd.print("Target:");
        lcd.print(inputNumber);

        lcd.setCursor(0, 1);
        lcd.print("#=GO *=CLEAR");
      }
    }

    // Confirm target
    else if (key == '#')
    {
      if (inputNumber.length() > 0)
      {
        targetPosition = inputNumber.toInt();
        inputNumber = "";

        lcd.clear();
        lcd.setCursor(0, 0);
        lcd.print("Target:");
        lcd.print(targetPosition);

        moveToTarget();
      }
    }

    // Clear target input
    else if (key == '*')
    {
      inputNumber = "";
      motorStop();
      showTarget();
    }

    // Zero current position
    else if (key == 'A')
    {
      motorStop();
      currentPosition = 0;

      lcd.clear();
      lcd.setCursor(0, 0);
      lcd.print("Position ZERO");

      lcd.setCursor(0, 1);
      lcd.print("Current: 0");

      delay(1200);
      showTarget();
    }

    // Stop motor
    else if (key == 'B')
    {
      motorStop();

      lcd.clear();
      lcd.setCursor(0, 0);
      lcd.print("MOTOR STOPPED");

      lcd.setCursor(0, 1);
      lcd.print("Current:");
      lcd.print(currentPosition);

      delay(1200);
      showTarget();
    }

    // Show current position
    else if (key == 'C')
    {
      motorStop();

      lcd.clear();
      lcd.setCursor(0, 0);
      lcd.print("Current:");
      lcd.print(currentPosition);

      lcd.setCursor(0, 1);
      lcd.print("Target:");
      lcd.print(targetPosition);

      delay(1200);
      showTarget();
    }

    // Reset position counter
    else if (key == 'D')
    {
      motorStop();
      currentPosition = 0;

      lcd.clear();
      lcd.setCursor(0, 0);
      lcd.print("COUNTER RESET");

      lcd.setCursor(0, 1);
      lcd.print("Current: 0");

      delay(1200);
      showTarget();
    }
  }
}

void moveToTarget()
{
  if (targetPosition == currentPosition)
  {
    motorStop();

    lcd.clear();
    lcd.setCursor(0, 0);
    lcd.print("TARGET REACHED");

    lcd.setCursor(0, 1);
    lcd.print("Pos:");
    lcd.print(currentPosition);

    delay(1200);
    showTarget();
    return;
  }

  // Move forward
  if (targetPosition > currentPosition)
  {
    motorForward();

    while (currentPosition < targetPosition)
    {
      // Simulation step:
      // Increase one position unit per step.
      currentPosition++;

      lcd.setCursor(0, 0);
      lcd.print("Moving Forward ");

      lcd.setCursor(0, 1);
      lcd.print("Pos:");
      lcd.print(currentPosition);
      lcd.print(" ");

      delay(100);
    }
  }

  // Move reverse
  else
  {
    motorReverse();

    while (currentPosition > targetPosition)
    {
      // Simulation step:
      // Decrease one position unit per step.
      currentPosition--;

      lcd.setCursor(0, 0);
      lcd.print("Moving Reverse ");

      lcd.setCursor(0, 1);
      lcd.print("Pos:");
      lcd.print(currentPosition);
      lcd.print(" ");

      delay(100);
    }
  }

  motorStop();

  lcd.clear();
  lcd.setCursor(0, 0);
  lcd.print("TARGET REACHED");

  lcd.setCursor(0, 1);
  lcd.print("Pos:");
  lcd.print(currentPosition);

  delay(1200);

  showTarget();
}

void motorForward()
{
  digitalWrite(EN1, HIGH);
  digitalWrite(IN1, HIGH);
  digitalWrite(IN2, LOW);
}

void motorReverse()
{
  digitalWrite(EN1, HIGH);
  digitalWrite(IN1, LOW);
  digitalWrite(IN2, HIGH);
}

void motorStop()
{
  digitalWrite(EN1, LOW);
  digitalWrite(IN1, LOW);
  digitalWrite(IN2, LOW);
}

void showTarget()
{
  lcd.clear();

  lcd.setCursor(0, 0);
  lcd.print("Target:");
  lcd.print(targetPosition);

  lcd.setCursor(0, 1);
  lcd.print("Current:");
  lcd.print(currentPosition);
}

Expected Output

The user enters a numerical position and confirms it with #. Arduino then drives the motor forward or reverse until the simulated current position reaches the selected target.

Target > Current: Motor moves forward
Target < Current: Motor moves reverse
Target = Current: Motor stops and shows TARGET REACHED
LCD: Displays current and target positions

Result

The Stepper Motor Position Selector experiment was implemented in Tinkercad using Arduino UNO, a 4×4 keypad, 16×2 I²C LCD and L293D motor driver. The entered target position controls the simulated motor movement and position display.
