Smart Medicine Reminder Box

← Arduino Experiments Aim Components Connections Working Procedure Code Result

Aim

To create a smart medicine reminder box using Arduino UNO, a 1-digit common-anode 7-segment display, two push buttons, a servo motor and a buzzer.

About the Experiment

The system starts with the medicine box closed and the display showing 5. Pressing the Start button begins a 5-second countdown on the 7-segment display.

When the timer reaches 0, the servo opens the medicine compartment to 90° and the buzzer sounds continuously. Pressing the Close button acknowledges the reminder, stops the buzzer, closes the box and resets the timer to 5.

Procedure

  1. Connect display segments A to G through 220Ω resistors to Arduino D2–D8.
  2. Connect the common anode of the display to 5V.
  3. Connect the servo signal wire to D9.
  4. Connect servo VCC to 5V and servo GND to GND.
  5. Connect buzzer positive to D10 and buzzer negative to GND.
  6. Connect one side of the Start button to A0 and the opposite side to GND.
  7. Connect one side of the Close button to A1 and the opposite side to GND.
  8. Connect Arduino UNO to the computer using the USB cable.
  9. Open Arduino IDE.
  10. Enter the program given below.
  11. Select Arduino UNO and the correct COM port.
  12. Verify and upload the program.
  13. Open Serial Monitor at 9600 baud.
  14. Press the Start button to begin the 5-second countdown.
  15. Observe the servo and buzzer when the timer reaches 0.
  16. Press the Close button to acknowledge the reminder and reset the system.

Arduino Code

Smart_Medicine_Reminder_Box.ino 
#include <Servo.h>

#define SEG_A 2
#define SEG_B 3
#define SEG_C 4
#define SEG_D 5
#define SEG_E 6
#define SEG_F 7
#define SEG_G 8

#define SERVO_PIN 9
#define BUZZER_PIN 10

#define BUTTON_START A0
#define BUTTON_CLOSE A1

#define CLOSED_POSITION 0
#define OPEN_POSITION 90

Servo medicineServo;

const int TIMER_START = 5;

int remainingTime = TIMER_START;

bool timerRunning = false;
bool boxOpen = false;

unsigned long lastSecond = 0;

void displayNumber(int number)
{
  digitalWrite(SEG_A, HIGH);
  digitalWrite(SEG_B, HIGH);
  digitalWrite(SEG_C, HIGH);
  digitalWrite(SEG_D, HIGH);
  digitalWrite(SEG_E, HIGH);
  digitalWrite(SEG_F, HIGH);
  digitalWrite(SEG_G, HIGH);

  switch (number)
  {
    case 0:
      digitalWrite(SEG_A, LOW);
      digitalWrite(SEG_B, LOW);
      digitalWrite(SEG_C, LOW);
      digitalWrite(SEG_D, LOW);
      digitalWrite(SEG_E, LOW);
      digitalWrite(SEG_F, LOW);
      break;

    case 1:
      digitalWrite(SEG_B, LOW);
      digitalWrite(SEG_C, LOW);
      break;

    case 2:
      digitalWrite(SEG_A, LOW);
      digitalWrite(SEG_B, LOW);
      digitalWrite(SEG_D, LOW);
      digitalWrite(SEG_E, LOW);
      digitalWrite(SEG_G, LOW);
      break;

    case 3:
      digitalWrite(SEG_A, LOW);
      digitalWrite(SEG_B, LOW);
      digitalWrite(SEG_C, LOW);
      digitalWrite(SEG_D, LOW);
      digitalWrite(SEG_G, LOW);
      break;

    case 4:
      digitalWrite(SEG_B, LOW);
      digitalWrite(SEG_C, LOW);
      digitalWrite(SEG_F, LOW);
      digitalWrite(SEG_G, LOW);
      break;

    case 5:
      digitalWrite(SEG_A, LOW);
      digitalWrite(SEG_C, LOW);
      digitalWrite(SEG_D, LOW);
      digitalWrite(SEG_F, LOW);
      digitalWrite(SEG_G, LOW);
      break;

    case 6:
      digitalWrite(SEG_A, LOW);
      digitalWrite(SEG_C, LOW);
      digitalWrite(SEG_D, LOW);
      digitalWrite(SEG_E, LOW);
      digitalWrite(SEG_F, LOW);
      digitalWrite(SEG_G, LOW);
      break;

    case 7:
      digitalWrite(SEG_A, LOW);
      digitalWrite(SEG_B, LOW);
      digitalWrite(SEG_C, LOW);
      break;

    case 8:
      digitalWrite(SEG_A, LOW);
      digitalWrite(SEG_B, LOW);
      digitalWrite(SEG_C, LOW);
      digitalWrite(SEG_D, LOW);
      digitalWrite(SEG_E, LOW);
      digitalWrite(SEG_F, LOW);
      digitalWrite(SEG_G, LOW);
      break;

    case 9:
      digitalWrite(SEG_A, LOW);
      digitalWrite(SEG_B, LOW);
      digitalWrite(SEG_C, LOW);
      digitalWrite(SEG_D, LOW);
      digitalWrite(SEG_F, LOW);
      digitalWrite(SEG_G, LOW);
      break;
  }
}

void setup()
{
  pinMode(SEG_A, OUTPUT);
  pinMode(SEG_B, OUTPUT);
  pinMode(SEG_C, OUTPUT);
  pinMode(SEG_D, OUTPUT);
  pinMode(SEG_E, OUTPUT);
  pinMode(SEG_F, OUTPUT);
  pinMode(SEG_G, OUTPUT);

  pinMode(BUZZER_PIN, OUTPUT);

  pinMode(BUTTON_START, INPUT_PULLUP);
  pinMode(BUTTON_CLOSE, INPUT_PULLUP);

  medicineServo.attach(SERVO_PIN);

  medicineServo.write(CLOSED_POSITION);

  noTone(BUZZER_PIN);

  displayNumber(TIMER_START);

  Serial.begin(9600);

  Serial.println("================================");
  Serial.println(" SMART MEDICINE REMINDER BOX");
  Serial.println(" TIMER = 5 SECONDS");
  Serial.println("================================");
  Serial.println("Press Button 1 to start.");
}

void loop()
{
  displayNumber(remainingTime);

  if (digitalRead(BUTTON_START) == LOW)
  {
    delay(50);

    if (digitalRead(BUTTON_START) == LOW)
    {
      remainingTime = TIMER_START;
      timerRunning = true;
      boxOpen = false;
      lastSecond = millis();

      Serial.println("Timer Started!");

      while (digitalRead(BUTTON_START) == LOW)
      {
        displayNumber(remainingTime);
      }

      delay(200);
    }
  }

  if (timerRunning && !boxOpen)
  {
    if (millis() - lastSecond >= 1000)
    {
      lastSecond = millis();

      if (remainingTime > 0)
      {
        remainingTime--;

        Serial.print("Remaining: ");
        Serial.print(remainingTime);
        Serial.println(" seconds");
      }

      if (remainingTime == 0)
      {
        timerRunning = false;
        medicineReminder();
      }
    }
  }

  if (digitalRead(BUTTON_CLOSE) == LOW)
  {
    delay(50);

    if (digitalRead(BUTTON_CLOSE) == LOW)
    {
      closeMedicineBox();

      while (digitalRead(BUTTON_CLOSE) == LOW)
      {
        displayNumber(5);
      }

      delay(200);
    }
  }
}

void medicineReminder()
{
  boxOpen = true;

  Serial.println();
  Serial.println("******************************");
  Serial.println(" MEDICATION TIME!");
  Serial.println(" BUZZER ON");
  Serial.println(" BOX OPEN");
  Serial.println("******************************");

  medicineServo.write(OPEN_POSITION);

  displayNumber(0);

  tone(BUZZER_PIN, 2000);

  while (digitalRead(BUTTON_CLOSE) == HIGH)
  {
    displayNumber(0);
  }

  noTone(BUZZER_PIN);

  delay(50);

  Serial.println("Medicine acknowledged.");

  closeMedicineBox();
}

void closeMedicineBox()
{
  if (boxOpen)
  {
    Serial.println("Closing medicine box...");

    medicineServo.write(CLOSED_POSITION);

    delay(1000);

    boxOpen = false;

    remainingTime = TIMER_START;
    timerRunning = false;

    displayNumber(TIMER_START);

    Serial.println("Box CLOSED.");
    Serial.println("Display = 5");
    Serial.println("Press Button 1 for next reminder.");
  }
}

Expected Output

The display initially shows 5. Pressing the Start button begins the countdown. When the display reaches 0, the servo opens the medicine box and the buzzer sounds. Pressing the Close button stops the buzzer, closes the box and resets the display to 5.

Initial Display: 5
Countdown: 5 → 4 → 3 → 2 → 1 → 0
At 0: Servo opens to 90° and buzzer turns ON
Close Button: Buzzer OFF, servo returns to 0°, display resets to 5

Result

The Smart Medicine Reminder Box experiment was implemented using Arduino UNO, a common-anode 7-segment display, two push buttons, a servo motor and a buzzer. The system counts down from 5 seconds, opens the medicine compartment and sounds an alarm when the reminder time is reached.
